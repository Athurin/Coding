注意是从后往前填充。先扩容在填充。

```cpp
#include <iostream>
 
using namespace std;
 
int main()
{
    string s;
    cin>>s;
 
    int oldsize = s.size()-1;
    int cnt = 0;
    for(int i=0; i<s.size(); i++)
    {
        if(s[i]>='0' && s[i]<='9')
            cnt++;
    }
 
    s.resize(s.size() + cnt*5);
    int newIndex = s.size()-1;
    int oldIndex = oldsize;
    while(oldIndex >= 0)
    {
        if(s[oldIndex]>='0' && s[oldIndex]<='9')
        {
            s[newIndex--]='r';
            s[newIndex--]='e';
            s[newIndex--]='b';
            s[newIndex--]='m';
            s[newIndex--]='u';
            s[newIndex--]='n';
        }
        else s[newIndex--]=s[oldIndex];
        oldIndex--;
    }
    cout<<s<<endl;
}
```
