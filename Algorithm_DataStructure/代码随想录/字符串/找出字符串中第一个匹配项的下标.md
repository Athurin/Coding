[28. 找出字符串中第一个匹配项的下标 - 力扣（LeetCode）](https://leetcode.cn/problems/find-the-index-of-the-first-occurrence-in-a-string/description/)

典型的KMP算法，事实证明，教材的代码时健全的。[[KMP]]：
朴素版：
```cpp
class Solution 
{
public:
    //朴素版的kmp算法
    //next数组
    void getNext(vector<int>& next, const string& needle)
    {
        int k=-1, j=0;
        next[0] = -1;
        while(j < needle.size()-1)
        {
            if(k == -1 || needle[k] == needle[j])
            {
                k++;
                j++;
                next[j] = k;
            }
            else k = next[k];
        }
    }

    //kmp算法
    int strStr(string haystack, string needle) 
    {
        int lenhay = haystack.size();
        int lenndl = needle.size();
        int i=0, j=0;

        vector<int> next(lenndl);
        getNext(next, needle);

        while(i < lenhay && j < lenndl)
        {
            if(j==-1 || haystack[i] == needle[j])
            {
                i++;
                j++;
            }
            else j=next[j];
        }

        if(j >= lenndl) return i-lenndl;
        else return -1;
    }
};
```

优化版：
nextval 的核心思想  
**“如果跳转后的那个字符还是等于当前失配字符，那么这次比较必败，不如直接再往前跳。”**
pat——模式串
```
nextval[0] = −1  
nextval[j] =  
 |—— next[j],　　　　　　　　　若 pat[j] ≠ pat[next[j]]  
 |—— nextval[next[j]],　　　　若 pat[j] = pat[next[j]]
```

| j           | 0  | 1  | 2  | 3  | 4 |
| ----------- | -- | -- | -- | -- | - |
| pat\[j]     | a  | a  | a  | a  | b |
| next\[j]    | -1 | 0  | 1  | 2  | 3 |
| nextval\[j] | -1 | -1 | -1 | -1 | 3 |

class Solution 
{
public:
    //朴素版的kmp算法
    //next数组
    void getNext(vector<int>& next, const string& needle)
    {
        int k=-1, j=0;
        next[0] = -1;
        while(j < needle.size()-1)
        {
            if(k == -1 || needle[k] == needle[j])
            {
                k++;
                j++;

                //next[j] = k;优化部分，跳过冗余比对
                if(needle[j] == needle[k]) //在模式串中碰到一样的就压缩信息
                {
                    next[j] = next[k];
                }
                else next[j] = k;

            }
            else k = next[k];
        }
    }

    //kmp算法
    int strStr(string haystack, string needle) 
    {
        int lenhay = haystack.size();
        int lenndl = needle.size();
        int i=0, j=0;

        vector<int> next(lenndl);
        getNext(next, needle);

        while(i < lenhay && j < lenndl)
        {
            if(j==-1 || haystack[i] == needle[j])
            {
                i++;
                j++;
            }
            else j=next[j];
        }

        if(j >= lenndl) return i-lenndl;
        else return -1;
    }
};
