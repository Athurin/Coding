
二叉树的递归遍历 也称 迭代法。模拟系统栈来完成的。

避免了递归层数太深而爆栈的问题。

# 前序遍历

[144. 二叉树的前序遍历 - 力扣（LeetCode）](https://leetcode.cn/problems/binary-tree-preorder-traversal/description/)

leetcode比较严格，其实在循环条件的时候判断不为空的时候就已经不是空指针了，无需判断cur是否为空。
当然，如果测试样例给到的根节点`root == nullptr` ，这样写就不多余。
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution 
{
public:
    //根左右的遍历，进栈顺序是“那么先将根节点放入栈中，然后将右孩子加入栈，再加入左孩子。”
    //每次出栈的都是当前子树的根节点，出栈的时候就做处理。
    //根节点出栈以后，再将改根节点的右子树入栈，然后是左子树入栈
    //下一次循环，左子树作为根节点先访问

    vector<int> preorderTraversal(TreeNode* root) 
    {
        vector<int> res;
        stack<TreeNode*> stk;
        stk.push(root);
        while(!stk.empty())
        {
            TreeNode* cur = stk.top();
            if(cur) res.push_back(cur->val);
            stk.pop();
            if(cur && cur->right) stk.push(cur->right);
            if(cur && cur->left)  stk.push(cur->left);
        }

        return res;
    }
};
```
如果测试样例给到的根节点`root == nullptr` ，也可以这样写，这样不用频繁判断`cur ?= nullptr`。
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution 
{
public:
    //根左右的遍历，进栈顺序是“那么先将根节点放入栈中，然后将右孩子加入栈，再加入左孩子。”
    //每次出栈的都是当前子树的根节点，出栈的时候就做处理。
    //根节点出栈以后，再将改根节点的右子树入栈，然后是左子树入栈
    //下一次循环，左子树作为根节点先访问

    vector<int> preorderTraversal(TreeNode* root) 
    {
        vector<int> res;
        if(!root) return res;

        stack<TreeNode*> stk;
        stk.push(root);
        while(!stk.empty())
        {
            TreeNode* cur = stk.top();
            res.push_back(cur->val);
            stk.pop();
            if(cur->right) stk.push(cur->right);
            if(cur->left)  stk.push(cur->left);
        }

        return res;
    }
};
```
