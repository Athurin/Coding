
二叉树的递归遍历 也称 迭代法。模拟系统栈来完成的。

避免了递归层数太深而爆栈的问题。

# 前序遍历

[144. 二叉树的前序遍历 - 力扣（LeetCode）](https://leetcode.cn/problems/binary-tree-preorder-traversal/description/)

leetcode比较严格，其实在循环条件的时候判断不为空的时候就已经不是空指针了，无需判断cur是否为空。
当然，如果测试样例给到的根节点`root == nullptr` ，这样写就不多余。
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution 
{
public:
    //根左右的遍历，进栈顺序是“那么先将根节点放入栈中，然后将右孩子加入栈，再加入左孩子。”
    //每次出栈的都是当前子树的根节点，出栈的时候就做处理。
    //根节点出栈以后，再将改根节点的右子树入栈，然后是左子树入栈
    //下一次循环，左子树作为根节点先访问

    vector<int> preorderTraversal(TreeNode* root) 
    {
        vector<int> res;
        stack<TreeNode*> stk;
        stk.push(root);
        while(!stk.empty())
        {
            TreeNode* cur = stk.top();
            if(cur) res.push_back(cur->val);
            stk.pop();
            if(cur && cur->right) stk.push(cur->right);
            if(cur && cur->left)  stk.push(cur->left);
        }

        return res;
    }
};
```

# 中序遍历

[94. 二叉树的中序遍历 - 力扣（LeetCode）](https://leetcode.cn/problems/binary-tree-inorder-traversal/description/)

```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution 
{
public:


    vector<int> inorderTraversal(TreeNode* root) 
    {
        vector<int> res;
        if(!root) return res;

        stack<TreeNode*> stk;
        TreeNode* cur = root;

        while(cur != nullptr || !stk.empty())
        {
            if(cur != nullptr)
            {
                stk.push(cur);
                cur = cur->left;
            }
            else
            {
                cur = stk.top();
                stk.pop();
                res.push_back(cur->val);
                cur = cur->right;
                //误点1，右子树的系欸但不需要在这里入栈，
                // cur = cur->right;  不要写成 stk.push(cur->right)
                //因为如果右节点不为空，在下一次循环会静茹if，压栈，然后向左下递归
                //如果右节点为空，表明没有右子树，写词循环将进入else处理栈顶

                //误点2， 不要加判空逻辑
                // cur = stk.top(); // 从栈里弹出的数据，就是要处理的数据（放进result数组里的数据）
                // stk.pop();
                // res.push_back(cur->val);     // 中
                // if(cur->right) cur = cur->right; //错误，内存超出限制
            }
        }

        return res;
    }
};
```

# 后序遍历

[145. 二叉树的后序遍历 - 力扣（LeetCode）](https://leetcode.cn/problems/binary-tree-postorder-traversal/description/)

再来看后序遍历，先序遍历是中左右，后序遍历是左右中，那么我们只需要调整一下先序遍历的代码顺序，就变成中右左的遍历顺序，然后在反转result数组，输出的结果顺序就是左右中了，如下图：

![前序到后序](https://file1.kamacoder.com/i/algo/20200808200338924.png)

**所以后序遍历只需要前序遍历的代码稍作修改就可以了，代码如下：**

```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution 
{
public:

    vector<int> postorderTraversal(TreeNode* root) 
    {
        vector<int> res;
        if(!root) return res;

        stack<TreeNode*> stk;
        stk.push(root);
        while(!stk.empty())
        {
            TreeNode* cur = stk.top();
            stk.pop();
            res.push_back(cur->val);
            if(cur->left) stk.push(cur->left);
            if(cur->right) stk.push(cur->right); //后进先出
        }
        reverse(res.begin(), res.end());

        return res;
    }
};
```
