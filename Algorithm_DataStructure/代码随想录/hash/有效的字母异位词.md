[242. 有效的字母异位词 - 力扣（LeetCode）](https://leetcode.cn/problems/valid-anagram/submissions/594140164/)

```cpp
class Solution 
{
public:
    bool isAnagram(string s, string t) 
    {
        int record[26] = {0};
        for (int i = 0; i < s.size(); i++) 
        {
            record[s[i] - 'a']++;
        }
        for (int i = 0; i < t.size(); i++) 
        {
            record[t[i] - 'a']--;
        }
        for (int i = 0; i < 26; i++) 
        {
            if (record[i] != 0) 
            {
                return false;
            }
        }
        // record数组所有元素都为零0，说明字符串s和t是字母异位词
        return true;
    }
};
```
