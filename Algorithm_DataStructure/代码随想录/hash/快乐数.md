[202. 快乐数 - 力扣（LeetCode）](https://leetcode.cn/problems/happy-number/submissions/654391100/)

AC:
```cpp
class Solution 
{
public:
    int getSum(int n)
    {
        int sum = 0;
        while(n)
        {
            sum += (n%10)*(n%10);
            n /= 10;
        }
        return sum;
    }

    bool isHappy(int n) 
    {
        //陷入无限循环的条件是sum重复出现
        //所以使用哈希表来记录sum是否会重复出现
        unordered_set<int> set;
        while(1)
        {
            int sum = getSum(n);
            if(sum == 1) return true;

            if(set.find(sum) != set.end()) return false;
            else
            {
                set.insert(sum);
                n = sum;
            }
        }
    }
};
```
